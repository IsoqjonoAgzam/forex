//@version=5
indicator("SMC All-In-One [Mr. Casino]", overlay=true, max_labels_count=500, max_boxes_on_chart=500, max_lines_count=500)

// ============================================================================
// Smart Money Concepts - All In One Indicator
// Based on Mr. Casino's Books
//
// Combines:
// 1. FU Candle Detection (SFU, AFU, 3 Candle FU)
// 2. Liquidity Zones (Equal Highs/Lows, Doji, Unfilled Wicks)
// 3. HCS Market Structure (Higher/Lower Close Structure + Negation)
// 4. Order Blocks (Bullish/Bearish OB)
//
// Trading Rules (from Mr. Casino):
// - Only trade in direction of HTF (4H/Daily) trend
// - Liquidity must be confirmed before entry
// - Risk:Reward minimum 1:3
// - Max risk per trade: 1-3%
// ============================================================================

// =================== INPUTS ===================
grpGeneral = "General Settings"
htfFilter       = input.bool(true, "Enable HTF Trend Filter", group=grpGeneral)
htfTimeframe    = input.timeframe("240", "HTF Timeframe", group=grpGeneral)
maxRisk         = input.float(2.0, "Max Risk Per Trade (%)", minval=0.5, maxval=5.0, group=grpGeneral)
minRR           = input.float(3.0, "Minimum Risk:Reward", minval=1.0, maxval=10.0, group=grpGeneral)

grpFU = "FU Candle Settings"
showFU          = input.bool(true, "Enable FU Detection", group=grpFU)
fuLookback      = input.int(20, "FU Lookback (Key Levels)", minval=5, maxval=100, group=grpFU)

grpLiq = "Liquidity Settings"
showLiquidity   = input.bool(true, "Enable Liquidity Zones", group=grpLiq)
eqTolerance     = input.float(0.05, "Equal Level Tolerance (%)", minval=0.01, maxval=1.0, group=grpLiq)
eqLookback      = input.int(3, "Min Candles for Equal Level", minval=2, maxval=10, group=grpLiq)

grpHCS = "HCS Structure Settings"
showHCS         = input.bool(true, "Enable HCS Structure", group=grpHCS)
hcsLength       = input.int(3, "HCS Consecutive Closes", minval=2, maxval=10, group=grpHCS)

grpOB = "Order Block Settings"
showOB          = input.bool(true, "Enable Order Blocks", group=grpOB)
obStrength      = input.int(3, "OB Strength (impulse candles)", minval=1, maxval=10, group=grpOB)
obImpulseMult   = input.float(1.5, "Impulse Size Multiplier", minval=1.0, maxval=5.0, group=grpOB)
maxOBAge        = input.int(100, "Max OB Age (bars)", minval=20, maxval=500, group=grpOB)

// =================== HELPER FUNCTIONS ===================
bodySize(idx) =>
    math.abs(close[idx] - open[idx])

candleRange(idx) =>
    high[idx] - low[idx]

upperWick(idx) =>
    high[idx] - math.max(close[idx], open[idx])

lowerWick(idx) =>
    math.min(close[idx], open[idx]) - low[idx]

isBullish(idx) =>
    close[idx] > open[idx]

isBearish(idx) =>
    close[idx] < open[idx]

isBIW_Bearish(idx) =>
    uWick = upperWick(idx)
    body = bodySize(idx)
    uWick > 0 and body <= uWick and isBearish(idx)

isBIW_Bullish(idx) =>
    lWick = lowerWick(idx)
    body = bodySize(idx)
    lWick > 0 and body <= lWick and isBullish(idx)

avgBody = ta.sma(bodySize(0), 20)

// =================== HTF TREND FILTER ===================
htfClose = request.security(syminfo.tickerid, htfTimeframe, close)
htfEMA20 = request.security(syminfo.tickerid, htfTimeframe, ta.ema(close, 20))
htfBullish = htfClose > htfEMA20
htfBearish = htfClose < htfEMA20

// =================== FU CANDLE DETECTION ===================
swingHigh = ta.pivothigh(high, fuLookback / 2, fuLookback / 2)
swingLow  = ta.pivotlow(low, fuLookback / 2, fuLookback / 2)

var float recentSwingHigh = na
var float recentSwingLow  = na

if not na(swingHigh)
    recentSwingHigh := swingHigh
if not na(swingLow)
    recentSwingLow := swingLow

// Liquidity sweeps
bearishSweep = not na(recentSwingHigh) and high > recentSwingHigh and close < recentSwingHigh
bullishSweep = not na(recentSwingLow) and low < recentSwingLow and close > recentSwingLow

// SFU (Strong FU with BIW)
bearishSFU = showFU and bearishSweep and isBIW_Bearish(0)
bullishSFU = showFU and bullishSweep and isBIW_Bullish(0)

// Apply HTF filter
bearishSFU := bearishSFU and (not htfFilter or htfBearish)
bullishSFU := bullishSFU and (not htfFilter or htfBullish)

// AFU (Attempted FU)
bearishAFU = showFU and bearishSweep and isBearish(0) and not isBIW_Bearish(0) and upperWick(0) > bodySize(0) * 0.5
bullishAFU = showFU and bullishSweep and isBullish(0) and not isBIW_Bullish(0) and lowerWick(0) > bodySize(0) * 0.5

bearishAFU := bearishAFU and (not htfFilter or htfBearish)
bullishAFU := bullishAFU and (not htfFilter or htfBullish)

// 3 Candle FU
tolerance = close * 0.01 / 100
bearish3CFU = showFU and not na(recentSwingHigh) and
     high[2] <= recentSwingHigh + tolerance and high[2] >= recentSwingHigh - tolerance and
     high[1] > recentSwingHigh and isBearish(0) and bodySize(0) > candleRange(0) * 0.5

bullish3CFU = showFU and not na(recentSwingLow) and
     low[2] >= recentSwingLow - tolerance and low[2] <= recentSwingLow + tolerance and
     low[1] < recentSwingLow and isBullish(0) and bodySize(0) > candleRange(0) * 0.5

// =================== LIQUIDITY ZONES ===================
isEqualHigh() =>
    tol = high * eqTolerance / 100
    count = 0
    refHigh = high
    for i = 1 to eqLookback
        if math.abs(high[i] - refHigh) <= tol
            count += 1
    count >= eqLookback - 1

isEqualLow() =>
    tol = low * eqTolerance / 100
    count = 0
    refLow = low
    for i = 1 to eqLookback
        if math.abs(low[i] - refLow) <= tol
            count += 1
    count >= eqLookback - 1

eqHighDetected = showLiquidity and isEqualHigh()
eqLowDetected  = showLiquidity and isEqualLow()

// =================== HCS STRUCTURE ===================
higherCloses() =>
    count = 0
    for i = 0 to hcsLength - 2
        if close[i] > close[i + 1]
            count += 1
        else
            break
    count

lowerCloses() =>
    count = 0
    for i = 0 to hcsLength - 2
        if close[i] < close[i + 1]
            count += 1
        else
            break
    count

bullishHCS = showHCS and higherCloses() >= hcsLength - 1
bearishHCS = showHCS and lowerCloses() >= hcsLength - 1

var int currentStructure = 0
var float structureHigh = na
var float structureLow = na

if bullishHCS
    currentStructure := 1
    structureLow := low[hcsLength - 1]

if bearishHCS
    currentStructure := -1
    structureHigh := high[hcsLength - 1]

bullishNegation = showHCS and currentStructure == 1 and not na(structureLow) and close < structureLow
bearishNegation = showHCS and currentStructure == -1 and not na(structureHigh) and close > structureHigh

if bullishNegation
    currentStructure := 0
    structureLow := na

if bearishNegation
    currentStructure := 0
    structureHigh := na

// =================== ORDER BLOCKS ===================
bullishImpulse() =>
    impulseSize = 0.0
    for i = 0 to obStrength - 1
        if isBullish(i)
            impulseSize += bodySize(i)
        else
            impulseSize := 0
            break
    impulseSize > avgBody * obImpulseMult * obStrength

bearishImpulse() =>
    impulseSize = 0.0
    for i = 0 to obStrength - 1
        if isBearish(i)
            impulseSize += bodySize(i)
        else
            impulseSize := 0
            break
    impulseSize > avgBody * obImpulseMult * obStrength

bullOBDetected = showOB and isBearish(obStrength) and bullishImpulse()
bearOBDetected = showOB and isBullish(obStrength) and bearishImpulse()

bullOBDetected := bullOBDetected and (not htfFilter or htfBullish)
bearOBDetected := bearOBDetected and (not htfFilter or htfBearish)

// OB Tracking
var float[] bullOBTop = array.new_float(0)
var float[] bullOBBot = array.new_float(0)
var int[] bullOBBar = array.new_int(0)
var box[] bullOBBox = array.new_box(0)

var float[] bearOBTop = array.new_float(0)
var float[] bearOBBot = array.new_float(0)
var int[] bearOBBar = array.new_int(0)
var box[] bearOBBox = array.new_box(0)

if bullOBDetected
    obTop = high[obStrength]
    obBot = low[obStrength]
    obBarIdx = bar_index - obStrength
    newBox = box.new(obBarIdx, obTop, bar_index + 20, obBot, 
         border_color=color.new(color.green, 50), bgcolor=color.new(color.green, 85))
    array.push(bullOBTop, obTop)
    array.push(bullOBBot, obBot)
    array.push(bullOBBar, obBarIdx)
    array.push(bullOBBox, newBox)

if bearOBDetected
    obTop = high[obStrength]
    obBot = low[obStrength]
    obBarIdx = bar_index - obStrength
    newBox = box.new(obBarIdx, obTop, bar_index + 20, obBot, 
         border_color=color.new(color.red, 50), bgcolor=color.new(color.red, 85))
    array.push(bearOBTop, obTop)
    array.push(bearOBBot, obBot)
    array.push(bearOBBar, obBarIdx)
    array.push(bearOBBox, newBox)

// Manage Bullish OBs
if array.size(bullOBTop) > 0
    for i = array.size(bullOBTop) - 1 to 0
        if i < array.size(bullOBTop)
            mitigated = close < array.get(bullOBBot, i)
            tooOld = (bar_index - array.get(bullOBBar, i)) > maxOBAge
            if mitigated or tooOld
                box.delete(array.get(bullOBBox, i))
                array.remove(bullOBTop, i)
                array.remove(bullOBBot, i)
                array.remove(bullOBBar, i)
                array.remove(bullOBBox, i)
            else
                box.set_right(array.get(bullOBBox, i), bar_index + 5)

// Manage Bearish OBs
if array.size(bearOBTop) > 0
    for i = array.size(bearOBTop) - 1 to 0
        if i < array.size(bearOBTop)
            mitigated = close > array.get(bearOBTop, i)
            tooOld = (bar_index - array.get(bearOBBar, i)) > maxOBAge
            if mitigated or tooOld
                box.delete(array.get(bearOBBox, i))
                array.remove(bearOBTop, i)
                array.remove(bearOBBot, i)
                array.remove(bearOBBar, i)
                array.remove(bearOBBox, i)
            else
                box.set_right(array.get(bearOBBox, i), bar_index + 5)

// =================== LABELS ===================
// FU Labels
if bearishSFU
    label.new(bar_index, high, "SFU", style=label.style_label_down, 
         color=color.red, textcolor=color.white, size=size.small)
if bullishSFU
    label.new(bar_index, low, "SFU", style=label.style_label_up, 
         color=color.green, textcolor=color.white, size=size.small)
if bearishAFU
    label.new(bar_index, high, "AFU", style=label.style_label_down, 
         color=color.new(color.red, 40), textcolor=color.white, size=size.tiny)
if bullishAFU
    label.new(bar_index, low, "AFU", style=label.style_label_up, 
         color=color.new(color.green, 40), textcolor=color.white, size=size.tiny)
if bearish3CFU
    label.new(bar_index, high, "3C-FU", style=label.style_label_down, 
         color=color.orange, textcolor=color.white, size=size.small)
if bullish3CFU
    label.new(bar_index, low, "3C-FU", style=label.style_label_up, 
         color=color.teal, textcolor=color.white, size=size.small)

// Liquidity Labels
if eqHighDetected
    line.new(bar_index - eqLookback, high, bar_index + 50, high, 
         color=color.new(color.red, 30), style=line.style_dashed, width=1)
    label.new(bar_index, high, "EQH", style=label.style_label_down, 
         color=color.new(color.red, 30), textcolor=color.white, size=size.tiny)
if eqLowDetected
    line.new(bar_index - eqLookback, low, bar_index + 50, low, 
         color=color.new(color.green, 30), style=line.style_dashed, width=1)
    label.new(bar_index, low, "EQL", style=label.style_label_up, 
         color=color.new(color.green, 30), textcolor=color.white, size=size.tiny)

// HCS Negation Labels
if bullishNegation
    label.new(bar_index, high, "BOS", style=label.style_label_down, 
         color=color.yellow, textcolor=color.black, size=size.small)
if bearishNegation
    label.new(bar_index, low, "BOS", style=label.style_label_up, 
         color=color.yellow, textcolor=color.black, size=size.small)

// OB Labels
if bullOBDetected
    label.new(bar_index - obStrength, low[obStrength], "OB", style=label.style_label_up, 
         color=color.new(color.green, 20), textcolor=color.white, size=size.tiny)
if bearOBDetected
    label.new(bar_index - obStrength, high[obStrength], "OB", style=label.style_label_down, 
         color=color.new(color.red, 20), textcolor=color.white, size=size.tiny)

// =================== INFO TABLE ===================
var table infoTable = table.new(position.top_right, 2, 6, bgcolor=color.new(color.black, 80), border_width=1)
if barstate.islast
    htfText = htfBullish ? "BULLISH" : htfBearish ? "BEARISH" : "NEUTRAL"
    htfCol = htfBullish ? color.green : htfBearish ? color.red : color.gray
    structText = currentStructure == 1 ? "BULLISH HCS" : currentStructure == -1 ? "BEARISH HCS" : "NEUTRAL"
    structCol = currentStructure == 1 ? color.green : currentStructure == -1 ? color.red : color.gray
    
    table.cell(infoTable, 0, 0, "Mr. Casino SMC", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 0, "v1.0", text_color=color.gray, text_size=size.small)
    table.cell(infoTable, 0, 1, "HTF Bias:", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 1, htfText, text_color=htfCol, text_size=size.small)
    table.cell(infoTable, 0, 2, "Structure:", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 2, structText, text_color=structCol, text_size=size.small)
    table.cell(infoTable, 0, 3, "Bull OBs:", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 3, str.tostring(array.size(bullOBTop)), text_color=color.green, text_size=size.small)
    table.cell(infoTable, 0, 4, "Bear OBs:", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 4, str.tostring(array.size(bearOBTop)), text_color=color.red, text_size=size.small)
    table.cell(infoTable, 0, 5, "Min RR:", text_color=color.white, text_size=size.small)
    table.cell(infoTable, 1, 5, "1:" + str.tostring(minRR, "#.#"), text_color=color.yellow, text_size=size.small)

// =================== ALERTS ===================
alertcondition(bearishSFU, "Bearish SFU", "Strong Bearish FU Candle Detected")
alertcondition(bullishSFU, "Bullish SFU", "Strong Bullish FU Candle Detected")
alertcondition(bullishNegation or bearishNegation, "Structure Break", "HCS Structure Break Detected")
alertcondition(bullOBDetected, "Bullish OB", "New Bullish Order Block")
alertcondition(bearOBDetected, "Bearish OB", "New Bearish Order Block")
alertcondition(eqHighDetected, "Equal Highs", "Equal Highs Liquidity Detected")
alertcondition(eqLowDetected, "Equal Lows", "Equal Lows Liquidity Detected")
