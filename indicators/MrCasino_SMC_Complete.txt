// =================== INPUTS ===================
grpGeneral = "General Settings"
htfFilter       = input.bool(true, "Enable HTF Trend Filter", group=grpGeneral)
htfTimeframe    = input.timeframe("240", "HTF Timeframe", group=grpGeneral)
rrRatio         = input.float(3.0, "Risk:Reward Ratio", minval=1.0, maxval=10.0, group=grpGeneral)
riskPct         = input.float(2.0, "Risk Per Trade (%)", minval=0.5, maxval=5.0, group=grpGeneral)
atrLength       = input.int(14, "ATR Length (for SL/TP)", minval=5, maxval=50, group=grpGeneral)
atrMult         = input.float(1.5, "ATR Multiplier for SL", minval=0.5, maxval=5.0, group=grpGeneral)

grpFU = "FU Candle Settings"
showFU          = input.bool(true, "Enable FU Detection", group=grpFU)
fuLookback      = input.int(20, "FU Lookback (Key Levels)", minval=5, maxval=100, group=grpFU)
show3CandleFU   = input.bool(true, "Show 3 Candle FU", group=grpFU)

grpOB = "Order Block Settings"
showOB          = input.bool(true, "Enable Order Blocks", group=grpOB)
obStrength      = input.int(3, "OB Strength (impulse candles)", minval=1, maxval=10, group=grpOB)
obImpulseMult   = input.float(1.5, "Impulse Size Multiplier", minval=1.0, maxval=5.0, group=grpOB)
maxOBAge        = input.int(100, "Max OB Age (bars)", minval=20, maxval=500, group=grpOB)

grpLiq = "Liquidity Settings"
showLiquidity   = input.bool(true, "Enable Liquidity Zones", group=grpLiq)
eqTolerance     = input.float(0.05, "Equal Level Tolerance (%)", minval=0.01, maxval=1.0, group=grpLiq)
eqLookback      = input.int(3, "Min Candles for Equal Level", minval=2, maxval=10, group=grpLiq)
showDoji        = input.bool(true, "Show Doji Targets", group=grpLiq)
showUnfilled    = input.bool(true, "Show Unfilled Wicks", group=grpLiq)
wickMultiplier  = input.float(2.0, "Wick Size Multiplier (vs Body)", minval=1.0, maxval=5.0, group=grpLiq)
dojiBodyPct     = input.float(10.0, "Doji Max Body (% of Range)", minval=1.0, maxval=30.0, group=grpLiq)

grpHCS = "HCS Structure Settings"
showHCS         = input.bool(true, "Enable HCS Structure", group=grpHCS)
hcsLength       = input.int(3, "HCS Consecutive Closes", minval=2, maxval=10, group=grpHCS)

grpDash = "Dashboard Settings"
showDashboard   = input.bool(true, "Show Dashboard Panel", group=grpDash)
dashPosition    = input.string("Top Right", "Dashboard Position", options=["Top Right", "Top Left", "Bottom Right", "Bottom Left"], group=grpDash)

// =================== HELPER FUNCTIONS ===================
bodySize(idx) =>
    math.abs(close[idx] - open[idx])

candleRange(idx) =>
    high[idx] - low[idx]

upperWick(idx) =>
    high[idx] - math.max(close[idx], open[idx])

lowerWick(idx) =>
    math.min(close[idx], open[idx]) - low[idx]

isBullish(idx) =>
    close[idx] > open[idx]

isBearish(idx) =>
    close[idx] < open[idx]

isBIW_Bearish(idx) =>
    uWick = upperWick(idx)
    body = bodySize(idx)
    uWick > 0 and body <= uWick and isBearish(idx)

isBIW_Bullish(idx) =>
    lWick = lowerWick(idx)
    body = bodySize(idx)
    lWick > 0 and body <= lWick and isBullish(idx)

avgBody = ta.sma(bodySize(0), 20)
atrVal = ta.atr(atrLength)

// =================== TIME TRACKING (1 WEEK = 7 DAYS) ===================
// Signals within last 7 days stay visible on chart (604800000 milliseconds)
oneWeekMs = 7 * 24 * 60 * 60 * 1000
isWithinOneWeek = (timenow - time) <= oneWeekMs

// =================== HTF TREND FILTER ===================
htfClose = request.security(syminfo.tickerid, htfTimeframe, close)
htfEMA20 = request.security(syminfo.tickerid, htfTimeframe, ta.ema(close, 20))
htfBullish = htfClose > htfEMA20
htfBearish = htfClose < htfEMA20

// =================== FU CANDLE DETECTION ===================
swingHigh = ta.pivothigh(high, fuLookback / 2, fuLookback / 2)
swingLow  = ta.pivotlow(low, fuLookback / 2, fuLookback / 2)

var float recentSwingHigh = na
var float recentSwingLow  = na

if not na(swingHigh)
    recentSwingHigh := swingHigh
if not na(swingLow)
    recentSwingLow := swingLow

// Liquidity Sweeps
bearishSweep = not na(recentSwingHigh) and high > recentSwingHigh and close < recentSwingHigh
bullishSweep = not na(recentSwingLow) and low < recentSwingLow and close > recentSwingLow

// SFU Detection (Strong FU - Body Inside Wick)
bearishSFU = showFU and bearishSweep and isBIW_Bearish(0)
bullishSFU = showFU and bullishSweep and isBIW_Bullish(0)
bearishSFU := bearishSFU and (not htfFilter or htfBearish)
bullishSFU := bullishSFU and (not htfFilter or htfBullish)

// AFU Detection (Attempted FU - weaker signal)
bearishAFU = showFU and bearishSweep and isBearish(0) and not isBIW_Bearish(0) and upperWick(0) > bodySize(0) * 0.5
bullishAFU = showFU and bullishSweep and isBullish(0) and not isBIW_Bullish(0) and lowerWick(0) > bodySize(0) * 0.5
bearishAFU := bearishAFU and (not htfFilter or htfBearish)
bullishAFU := bullishAFU and (not htfFilter or htfBullish)

// 3 Candle FU Detection
// Bearish: candle 1 approaches resistance, candle 2 breaks above, candle 3 reverses bearish
// Bullish: candle 1 approaches support, candle 2 breaks below, candle 3 reverses bullish
fuTolerance = close * 0.01 / 100
bearish3CFU = show3CandleFU and showFU and not na(recentSwingHigh) and
     high[2] <= recentSwingHigh + fuTolerance and high[2] >= recentSwingHigh - fuTolerance and
     high[1] > recentSwingHigh and
     isBearish(0) and bodySize(0) > candleRange(0) * 0.5
bearish3CFU := bearish3CFU and (not htfFilter or htfBearish)

bullish3CFU = show3CandleFU and showFU and not na(recentSwingLow) and
     low[2] >= recentSwingLow - fuTolerance and low[2] <= recentSwingLow + fuTolerance and
     low[1] < recentSwingLow and
     isBullish(0) and bodySize(0) > candleRange(0) * 0.5
bullish3CFU := bullish3CFU and (not htfFilter or htfBullish)

// =================== ORDER BLOCKS ===================
bullishImpulse() =>
    impulseSize = 0.0
    for i = 0 to obStrength - 1
        if isBullish(i)
            impulseSize += bodySize(i)
        else
            impulseSize := 0
            break
    impulseSize > avgBody * obImpulseMult * obStrength

bearishImpulse() =>
    impulseSize = 0.0
    for i = 0 to obStrength - 1
        if isBearish(i)
            impulseSize += bodySize(i)
        else
            impulseSize := 0
            break
    impulseSize > avgBody * obImpulseMult * obStrength

bullOBDetected = showOB and isBearish(obStrength) and bullishImpulse()
bearOBDetected = showOB and isBullish(obStrength) and bearishImpulse()
bullOBDetected := bullOBDetected and (not htfFilter or htfBullish)
bearOBDetected := bearOBDetected and (not htfFilter or htfBearish)

// OB Tracking with Mitigation
var float[] bullOBTop = array.new_float(0)
var float[] bullOBBot = array.new_float(0)
var int[] bullOBBar = array.new_int(0)
var box[] bullOBBox = array.new_box(0)

var float[] bearOBTop = array.new_float(0)
var float[] bearOBBot = array.new_float(0)
var int[] bearOBBar = array.new_int(0)
var box[] bearOBBox = array.new_box(0)

if bullOBDetected and isWithinOneWeek
    obTop = high[obStrength]
    obBot = low[obStrength]
    obBarIdx = bar_index - obStrength
    newBox = box.new(obBarIdx, obTop, bar_index + 20, obBot, 
         border_color=color.new(color.green, 50), bgcolor=color.new(color.green, 88))
    array.push(bullOBTop, obTop)
    array.push(bullOBBot, obBot)
    array.push(bullOBBar, obBarIdx)
    array.push(bullOBBox, newBox)

if bearOBDetected and isWithinOneWeek
    obTop = high[obStrength]
    obBot = low[obStrength]
    obBarIdx = bar_index - obStrength
    newBox = box.new(obBarIdx, obTop, bar_index + 20, obBot, 
         border_color=color.new(color.red, 50), bgcolor=color.new(color.red, 88))
    array.push(bearOBTop, obTop)
    array.push(bearOBBot, obBot)
    array.push(bearOBBar, obBarIdx)
    array.push(bearOBBox, newBox)

// Manage Bullish OBs (mitigation + age check)
if array.size(bullOBTop) > 0
    for i = array.size(bullOBTop) - 1 to 0
        if i < array.size(bullOBTop)
            mitigated = close < array.get(bullOBBot, i)
            tooOld = (bar_index - array.get(bullOBBar, i)) > maxOBAge
            if mitigated or tooOld
                box.delete(array.get(bullOBBox, i))
                array.remove(bullOBTop, i)
                array.remove(bullOBBot, i)
                array.remove(bullOBBar, i)
                array.remove(bullOBBox, i)
            else
                box.set_right(array.get(bullOBBox, i), bar_index + 5)

// Manage Bearish OBs (mitigation + age check)
if array.size(bearOBTop) > 0
    for i = array.size(bearOBTop) - 1 to 0
        if i < array.size(bearOBTop)
            mitigated = close > array.get(bearOBTop, i)
            tooOld = (bar_index - array.get(bearOBBar, i)) > maxOBAge
            if mitigated or tooOld
                box.delete(array.get(bearOBBox, i))
                array.remove(bearOBTop, i)
                array.remove(bearOBBot, i)
                array.remove(bearOBBar, i)
                array.remove(bearOBBox, i)
            else
                box.set_right(array.get(bearOBBox, i), bar_index + 5)

// =================== HCS MARKET STRUCTURE ===================
higherCloses() =>
    count = 0
    for i = 0 to hcsLength - 2
        if close[i] > close[i + 1]
            count += 1
        else
            break
    count

lowerCloses() =>
    count = 0
    for i = 0 to hcsLength - 2
        if close[i] < close[i + 1]
            count += 1
        else
            break
    count

bullishHCS = showHCS and higherCloses() >= hcsLength - 1
bearishHCS = showHCS and lowerCloses() >= hcsLength - 1

var int currentStructure = 0
var float structureHigh = na
var float structureLow = na

if bullishHCS
    currentStructure := 1
    structureLow := low[hcsLength - 1]

if bearishHCS
    currentStructure := -1
    structureHigh := high[hcsLength - 1]

// HCS Negation (Break of Structure)
bullishNegation = showHCS and currentStructure == 1 and not na(structureLow) and close < structureLow
bearishNegation = showHCS and currentStructure == -1 and not na(structureHigh) and close > structureHigh

if bullishNegation
    currentStructure := 0
    structureLow := na
if bearishNegation
    currentStructure := 0
    structureHigh := na

// =================== LIQUIDITY ZONES ===================
isEqualHigh() =>
    tol = high * eqTolerance / 100
    count = 0
    for i = 1 to eqLookback
        if math.abs(high[i] - high) <= tol
            count += 1
    count >= eqLookback - 1

isEqualLow() =>
    tol = low * eqTolerance / 100
    count = 0
    for i = 1 to eqLookback
        if math.abs(low[i] - low) <= tol
            count += 1
    count >= eqLookback - 1

eqHighDetected = showLiquidity and isEqualHigh()
eqLowDetected  = showLiquidity and isEqualLow()

// Doji Detection (manipulation target)
isDoji() =>
    range_val = candleRange(0)
    body = bodySize(0)
    range_val > 0 and (body / range_val * 100) <= dojiBodyPct

dojiDetected = showDoji and isDoji()

// Large Unfilled Wick Detection
hasLargeUpperWick() =>
    body = bodySize(0)
    uWick = upperWick(0)
    body > 0 and uWick >= body * wickMultiplier

hasLargeLowerWick() =>
    body = bodySize(0)
    lWick = lowerWick(0)
    body > 0 and lWick >= body * wickMultiplier

largeUpperWick = showUnfilled and hasLargeUpperWick()
largeLowerWick = showUnfilled and hasLargeLowerWick()

// =================== COMBINED ENTRY SIGNALS ===================
// Strong entry = SFU or 3C-FU confirmed by structure direction + OB
bullEntry = bullishSFU or bullish3CFU or (bullOBDetected and currentStructure == 1)
bearEntry = bearishSFU or bearish3CFU or (bearOBDetected and currentStructure == -1)

// Any signal (for counting)
anyBullSignal = bullishSFU or bullishAFU or bullish3CFU or bullOBDetected
anyBearSignal = bearishSFU or bearishAFU or bearish3CFU or bearOBDetected

// =================== SL/TP CALCULATION ===================
// Bull: SL below low, TP = entry + (entry - SL) * RR
bullSL = low - atrVal * atrMult
bullTP = close + (close - bullSL) * rrRatio

// Bear: SL above high, TP = entry - (SL - entry) * RR
bearSL = high + atrVal * atrMult
bearTP = close - (bearSL - close) * rrRatio

// =================== 1-WEEK PERFORMANCE TRACKING ===================
var int weekBullEntries = 0
var int weekBearEntries = 0
var int weekTPHit = 0
var int weekSLHit = 0
var float weekProfit = 0.0
var float weekLoss = 0.0

// Track active trades
var float[] activeBullEntry = array.new_float(0)
var float[] activeBullSL = array.new_float(0)
var float[] activeBullTP = array.new_float(0)
var int[] activeBullTime = array.new_int(0)

var float[] activeBearEntry = array.new_float(0)
var float[] activeBearSL = array.new_float(0)
var float[] activeBearTP = array.new_float(0)
var int[] activeBearTime = array.new_int(0)

// Reset weekly counters at start of new week
newWeek = ta.change(time("W")) != 0
if newWeek
    weekBullEntries := 0
    weekBearEntries := 0
    weekTPHit := 0
    weekSLHit := 0
    weekProfit := 0.0
    weekLoss := 0.0

// Record new bull entries
if bullEntry and isWithinOneWeek
    weekBullEntries += 1
    array.push(activeBullEntry, close)
    array.push(activeBullSL, bullSL)
    array.push(activeBullTP, bullTP)
    array.push(activeBullTime, time)

// Record new bear entries
if bearEntry and isWithinOneWeek
    weekBearEntries += 1
    array.push(activeBearEntry, close)
    array.push(activeBearSL, bearSL)
    array.push(activeBearTP, bearTP)
    array.push(activeBearTime, time)

// Check bull trades for TP/SL hit
if array.size(activeBullEntry) > 0
    for i = array.size(activeBullEntry) - 1 to 0
        if i < array.size(activeBullEntry)
            entryPrice = array.get(activeBullEntry, i)
            slPrice = array.get(activeBullSL, i)
            tpPrice = array.get(activeBullTP, i)
            tradeTime = array.get(activeBullTime, i)
            
            if (timenow - tradeTime) <= oneWeekMs
                if high >= tpPrice
                    weekTPHit += 1
                    weekProfit += (tpPrice - entryPrice) / entryPrice * 100
                    array.remove(activeBullEntry, i)
                    array.remove(activeBullSL, i)
                    array.remove(activeBullTP, i)
                    array.remove(activeBullTime, i)
                else if low <= slPrice
                    weekSLHit += 1
                    weekLoss += (entryPrice - slPrice) / entryPrice * 100
                    array.remove(activeBullEntry, i)
                    array.remove(activeBullSL, i)
                    array.remove(activeBullTP, i)
                    array.remove(activeBullTime, i)
            else
                array.remove(activeBullEntry, i)
                array.remove(activeBullSL, i)
                array.remove(activeBullTP, i)
                array.remove(activeBullTime, i)

// Check bear trades for TP/SL hit
if array.size(activeBearEntry) > 0
    for i = array.size(activeBearEntry) - 1 to 0
        if i < array.size(activeBearEntry)
            entryPrice = array.get(activeBearEntry, i)
            slPrice = array.get(activeBearSL, i)
            tpPrice = array.get(activeBearTP, i)
            tradeTime = array.get(activeBearTime, i)
            
            if (timenow - tradeTime) <= oneWeekMs
                if low <= tpPrice
                    weekTPHit += 1
                    weekProfit += (entryPrice - tpPrice) / entryPrice * 100
                    array.remove(activeBearEntry, i)
                    array.remove(activeBearSL, i)
                    array.remove(activeBearTP, i)
                    array.remove(activeBearTime, i)
                else if high >= slPrice
                    weekSLHit += 1
                    weekLoss += (slPrice - entryPrice) / entryPrice * 100
                    array.remove(activeBearEntry, i)
                    array.remove(activeBearSL, i)
                    array.remove(activeBearTP, i)
                    array.remove(activeBearTime, i)
            else
                array.remove(activeBearEntry, i)
                array.remove(activeBearSL, i)
                array.remove(activeBearTP, i)
                array.remove(activeBearTime, i)

// =================== SIGNAL LABELS (persist for 1 week) ===================
// All signals within last 7 days remain visible on the chart

// SFU Labels with SL/TP lines
if bullishSFU and isWithinOneWeek
    label.new(bar_index, low, "SFU\nBUY", style=label.style_label_up, 
         color=color.green, textcolor=color.white, size=size.small)
    line.new(bar_index, bullSL, bar_index + 10, bullSL, color=color.red, style=line.style_dotted, width=1)
    line.new(bar_index, bullTP, bar_index + 10, bullTP, color=color.green, style=line.style_dotted, width=1)

if bearishSFU and isWithinOneWeek
    label.new(bar_index, high, "SFU\nSELL", style=label.style_label_down, 
         color=color.red, textcolor=color.white, size=size.small)
    line.new(bar_index, bearSL, bar_index + 10, bearSL, color=color.red, style=line.style_dotted, width=1)
    line.new(bar_index, bearTP, bar_index + 10, bearTP, color=color.green, style=line.style_dotted, width=1)

// AFU Labels
if bullishAFU and isWithinOneWeek
    label.new(bar_index, low, "AFU\nBUY", style=label.style_label_up, 
         color=color.new(color.green, 30), textcolor=color.white, size=size.tiny)

if bearishAFU and isWithinOneWeek
    label.new(bar_index, high, "AFU\nSELL", style=label.style_label_down, 
         color=color.new(color.red, 30), textcolor=color.white, size=size.tiny)

// 3 Candle FU Labels
if bullish3CFU and isWithinOneWeek
    label.new(bar_index, low, "3C-FU\nBUY", style=label.style_label_up, 
         color=color.teal, textcolor=color.white, size=size.small)
    line.new(bar_index, bullSL, bar_index + 10, bullSL, color=color.red, style=line.style_dotted, width=1)
    line.new(bar_index, bullTP, bar_index + 10, bullTP, color=color.green, style=line.style_dotted, width=1)

if bearish3CFU and isWithinOneWeek
    label.new(bar_index, high, "3C-FU\nSELL", style=label.style_label_down, 
         color=color.orange, textcolor=color.white, size=size.small)
    line.new(bar_index, bearSL, bar_index + 10, bearSL, color=color.red, style=line.style_dotted, width=1)
    line.new(bar_index, bearTP, bar_index + 10, bearTP, color=color.green, style=line.style_dotted, width=1)

// Order Block Labels
if bullOBDetected and isWithinOneWeek
    label.new(bar_index - obStrength, low[obStrength], "OB\nBUY", style=label.style_label_up, 
         color=color.new(color.teal, 0), textcolor=color.white, size=size.small)

if bearOBDetected and isWithinOneWeek
    label.new(bar_index - obStrength, high[obStrength], "OB\nSELL", style=label.style_label_down, 
         color=color.new(color.orange, 0), textcolor=color.white, size=size.small)

// HCS Break of Structure Labels
if bullishNegation and isWithinOneWeek
    label.new(bar_index, high, "BOS\nUP", style=label.style_label_down, 
         color=color.yellow, textcolor=color.black, size=size.small)

if bearishNegation and isWithinOneWeek
    label.new(bar_index, low, "BOS\nDOWN", style=label.style_label_up, 
         color=color.yellow, textcolor=color.black, size=size.small)

// Equal Highs/Lows (Liquidity Pools)
if eqHighDetected and isWithinOneWeek
    line.new(bar_index - eqLookback, high, bar_index + 50, high, 
         color=color.new(color.red, 30), style=line.style_dashed, width=2)
    label.new(bar_index, high, "EQH\nSell Liq", style=label.style_label_down, 
         color=color.new(color.red, 30), textcolor=color.white, size=size.tiny)

if eqLowDetected and isWithinOneWeek
    line.new(bar_index - eqLookback, low, bar_index + 50, low, 
         color=color.new(color.green, 30), style=line.style_dashed, width=2)
    label.new(bar_index, low, "EQL\nBuy Liq", style=label.style_label_up, 
         color=color.new(color.green, 30), textcolor=color.white, size=size.tiny)

// Doji Detection (manipulation target)
if dojiDetected and isWithinOneWeek
    label.new(bar_index, high, "DOJI", style=label.style_diamond, 
         color=color.new(color.purple, 30), textcolor=color.white, size=size.tiny)

// Unfilled Wicks (zones price will return to fill)
if largeUpperWick and isWithinOneWeek
    bodyTop = math.max(close, open)
    box.new(bar_index, high, bar_index + 50, bodyTop, 
         border_color=color.new(color.red, 40), bgcolor=color.new(color.red, 90))
    label.new(bar_index, high, "UW", style=label.style_label_down, 
         color=color.new(color.red, 40), textcolor=color.white, size=size.tiny)

if largeLowerWick and isWithinOneWeek
    bodyBottom = math.min(close, open)
    box.new(bar_index, bodyBottom, bar_index + 50, low, 
         border_color=color.new(color.green, 40), bgcolor=color.new(color.green, 90))
    label.new(bar_index, low, "UW", style=label.style_label_up, 
         color=color.new(color.green, 40), textcolor=color.white, size=size.tiny)

// =================== PERFORMANCE DASHBOARD (CORNER PANEL) ===================
dashPos = dashPosition == "Top Right" ? position.top_right : 
     dashPosition == "Top Left" ? position.top_left : 
     dashPosition == "Bottom Right" ? position.bottom_right : position.bottom_left

var table dashboard = table.new(dashPos, 2, 14, 
     bgcolor=color.new(color.black, 75), border_width=1, border_color=color.new(color.white, 70))

if barstate.islast and showDashboard
    totalEntries = weekBullEntries + weekBearEntries
    totalClosed = weekTPHit + weekSLHit
    winRate = totalClosed > 0 ? weekTPHit / totalClosed * 100 : 0.0
    lossRate = totalClosed > 0 ? weekSLHit / totalClosed * 100 : 0.0
    netProfitPct = weekProfit - weekLoss
    activeTrades = array.size(activeBullEntry) + array.size(activeBearEntry)
    activeBullOBs = array.size(bullOBTop)
    activeBearOBs = array.size(bearOBTop)
    
    htfText = htfBullish ? "BULLISH" : htfBearish ? "BEARISH" : "NEUTRAL"
